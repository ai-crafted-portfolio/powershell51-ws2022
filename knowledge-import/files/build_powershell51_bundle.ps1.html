param(
  [string]$RecordsDir = '.\records',
  [string]$OutputPath = '.\powershell51_knowledge_bundle_cp932.txt'
)
$ErrorActionPreference = 'Stop'
$ExpectedRecords = 4030
$Encoding932 = [System.Text.Encoding]::GetEncoding(932)
$recordFiles = Get-ChildItem -LiteralPath $RecordsDir -Filter 'PS51-WS2022-2026-*.txt' -File | Sort-Object Name
if ($recordFiles.Count -ne $ExpectedRecords) {
  throw "records ?????? $ExpectedRecords ????????: $($recordFiles.Count)"
}
$builder = New-Object System.Text.StringBuilder
foreach ($recordFile in $recordFiles) {
  $text = Get-Content -LiteralPath $recordFile.FullName -Raw -Encoding UTF8
  if (-not $text.StartsWith('###KnowledgeNo###')) {
    throw "??????????: $($recordFile.Name)"
  }
  $normalized = $text -replace "`r`n", "`n" -replace "`r", "`n"
  $normalized = $normalized.TrimEnd("`n")
  [void]$builder.Append($normalized.Replace("`n", "`r`n"))
  [void]$builder.Append("`r`n")
}
$bundleText = $builder.ToString()
$bundleBytes = $Encoding932.GetBytes($bundleText)
[System.IO.File]::WriteAllBytes((Join-Path (Get-Location) $OutputPath), $bundleBytes)
Write-Host "BUILD PASS output=$OutputPath records=$($recordFiles.Count) encoding=CP932 line_endings=CRLF"
