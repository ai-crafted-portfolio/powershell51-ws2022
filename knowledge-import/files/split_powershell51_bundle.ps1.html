param(
  [string]$BundlePath = '.\powershell51_knowledge_bundle_cp932.txt',
  [string]$OutputDir = '.\records'
)
$ErrorActionPreference = 'Stop'
$ExpectedRecords = 4030
$Utf8NoBom = New-Object System.Text.UTF8Encoding($false)
$Encoding932 = [System.Text.Encoding]::GetEncoding(932)
$bundleText = $Encoding932.GetString([System.IO.File]::ReadAllBytes($BundlePath))
$records = [regex]::Split($bundleText, '(?=###KnowledgeNo###\r?\n)') | Where-Object { $_ }
if ($records.Count -ne $ExpectedRecords) {
  throw "bundle ??????????????: $ExpectedRecords ??: $($records.Count)"
}
New-Item -ItemType Directory -Path $OutputDir -Force | Out-Null
for ($index = 0; $index -lt $records.Count; $index++) {
  $recordNo = $index + 1
  $recordName = ('PS51-WS2022-2026-{0:D4}.txt' -f $recordNo)
  $normalized = $records[$index] -replace "`r`n", "`n" -replace "`r", "`n"
  $normalized = $normalized.TrimEnd("`n").Replace("`n", "`r`n") + "`r`n"
  [System.IO.File]::WriteAllText((Join-Path $OutputDir $recordName), $normalized, $Utf8NoBom)
}
Write-Host "SPLIT PASS output=$OutputDir records=$($records.Count) encoding=UTF-8"
